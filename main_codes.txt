const loginUser = async (req, res) => {
  try {
    const { email, password, username } = req.body;
    const identifier = (email || username || '').trim();

    if (!identifier || !password) {
      return res.status(400).json({ message: 'Please provide email/username and password' });
    }

    const identifierRegex = new RegExp(`^${identifier.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}$`, 'i');
    const user = await User.findOne({
      $or: [{ email: identifierRegex }, { username: identifierRegex }]
    });

    if (!user || !(await user.matchPassword(password))) {
      return res.status(401).json({ message: 'Invalid phone number or password.' });
    }

    return res.json({
      message: 'Login successful!',
      token: generateToken(user._id),
      user: {
        id: user._id,
        name: user.name,
        email: user.email,
        phone: user.phone,
        role: user.role
      }
    });
  } catch (error) {
    res.status(500).json({ message: 'Server error', error: error.message });
  }
};


Raise Complaints:
const createComplaint = async (req, res) => {
  try {
    const { parkId, parkName, locationInPark, category, priority, description, latitude, longitude, ward, remarks } = req.body;
    
    if (!latitude || !longitude || !ward) {
      return res.status(400).json({ message: 'Latitude, longitude, and ward are required' });
    }

    let imageName = null;
    if (req.files && req.files.length > 0) {
      imageName = req.files[0].filename || req.files[0].path;
    }

    const complaintNumber = 'CMP' + Math.floor(100000000 + Math.random() * 900000000);
    const complaint = await Complaint.create({
      complaintNumber,
      user: req.user ? req.user._id : req.body.userId,
      park: parkId,
      parkName,
      locationInPark,
      category: category || 'General Maintenance',
      priority: priority || 'Medium',
      description: description || remarks,
      latitude,
      longitude,
      ward,
      remarks,
      images: imageName ? [imageName] : [],
      status: 'New'
    });

    const complaintId = complaint.complaintNumber || complaint._id;
    const userEmail = req.user?.email || req.body.email;
    if (userEmail) {
      sendEmailNotification(userEmail, 'Complaint Registered - Reference ID', 
        `Your complaint has been registered successfully.\nComplaint ID: ${complaintId}`
      );
    }

    return res.status(201).json({
      success: true,
      message: `Complaint submitted successfully. Your Complaint ID is: ${complaintId}`,
      complaint
    });
  } catch (error) {
    res.status(500).json({ message: 'Server error', error: error.message });
  }
};


Assign Complaints:
const assignOfficials = async (req, res) => {
  try {
    const { complaintId, officialId } = req.body;
    if (!complaintId || !officialId) {
      return res.status(400).json({ message: 'Complaint ID and Official ID are required' });
    }

    const official = await User.findById(officialId);
    const complaint = await Complaint.findById(complaintId);
    if (!official || !complaint) {
      return res.status(404).json({ message: 'Official or Complaint not found' });
    }

    complaint.assignedOfficial = official._id;
    complaint.status = 'In progress';
    complaint.assignedAt = new Date();
    await complaint.save();

    if (official.email) {
      sendEmailNotification(official.email, 'New Complaint Assigned', `A new complaint #${complaint.complaintNumber} has been assigned to you.`);
    }

    return res.json({ success: true, message: 'Complaint assigned successfully and email sent.', complaint });
  } catch (error) {
    res.status(500).json({ message: 'Server error', error: error.message });
  }
};

const updateDivision = async (req, res) => {
  try {
    const { id } = req.params;
    const { div_name, zone_name, latitude, longitude } = req.body;
    if (!div_name || !zone_name || !latitude || !longitude) {
      return res.status(400).json({ message: 'All division fields are required' });
    }
    const division = await Corporation.findById(id);
    if (!division) {
      return res.status(404).json({ message: 'Division not found' });
    }
    division.name = div_name;
    division.zone_name = zone_name;
    division.latitude = latitude;
    division.longitude = longitude;
    await division.save();
    return res.json({ success: true, message: 'Division updated successfully!', division });
  } catch (error) {
    return res.status(400).json({ message: error.message });
  }
};

const sendOtp = async (req, res) => {
  try {
    const { email } = req.body;
    if (!email) {
      return res.status(400).json({ message: 'Email is required.' });
    }
    const user = await User.findOne({ email: new RegExp(`^${email}$`, 'i') });
    if (!user) {
      return res.status(404).json({ message: 'Email does not exist.' });
    }
    const otp = Math.floor(100000 + Math.random() * 900000).toString();
    user.resetPasswordOtp = otp;
    user.resetPasswordOtpExpires = new Date(Date.now() + 10 * 60 * 1000);
    await user.save();

    await sendPasswordResetOtpEmail(user.email, user.name, otp);
    return res.json({ success: true, message: 'OTP sent to email for password reset.', email: user.email });
  } catch (error) {
    return res.status(500).json({ message: 'Server error', error: error.message });
  }
};

const mongoose = require('mongoose');

const wardSchema = new mongoose.Schema({
  division_name: { type: String, ref: 'Corporation' },
  ward_name: { type: String, required: true },
  ward_no: { type: String, required: true },
  districtId: { type: mongoose.Schema.Types.ObjectId, ref: 'District' },
  zoneId: { type: mongoose.Schema.Types.ObjectId, ref: 'Zone' },
  status: { type: String, default: 'Active', enum: ['Active', 'Inactive'] }
}, { timestamps: true });

module.exports = mongoose.model('Ward', wardSchema);
